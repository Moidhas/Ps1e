lw $1 0($2) // MEM
addi $1 $1 10 // EXE
// Because of the forwarding unit, the EXE/MEM pipeline register will be prioritized over THE MEM/WB pipeline register.
// So the addi will use the value of $1 before it is changed by lw's.
